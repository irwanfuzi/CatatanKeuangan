(function() { 'use strict'; window.pdf2htmlEX = window.pdf2htmlEX || {}; function Viewer(config) { this.container = document.getElementById(config.container_id); this.sidebar = document.getElementById(config.sidebar_id); this.pages = this.container.getElementsByClassName('pf'); this.init(); } Viewer.prototype.init = function() { this.setupNavigation(); console.log("PDF Viewer Engine initialized successfully."); }; Viewer.prototype.setupNavigation = function() { const links = this.sidebar.querySelectorAll('a'); links.forEach(link => { link.addEventListener('click', (e) => { e.preventDefault(); const targetId = link.getAttribute('href').substring(1); const targetPage = document.getElementById(targetId); if (targetPage) { targetPage.scrollIntoView({ behavior: 'smooth' }); } }); }); }; window.pdf2htmlEX.Viewer = Viewer; })(); // Jalankan Script setelah DOM ter-load penuh document.addEventListener("DOMContentLoaded", function() { if (document.getElementById("page-container") && document.getElementById("sidebar")) { new window.pdf2htmlEX.Viewer({ container_id: "page-container", sidebar_id: "sidebar" }); } });
